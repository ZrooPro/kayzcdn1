const SECTION_TITLES = ['「 ESSENTIAL & CREATIVE 」', '「 COMMUNITY & ENTERTAINMENT 」', '「 BUSINESS & CONTROL 」']
const weatherCache = new Map()
let cachedBanner = null
const clean = value => String(value ?? '').replace(/[\x00-\x1f\x7f]/g, ' ').trim()
const escapeHtml = value => String(value ?? '').replace(/[&<>"']/g, ch => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[ch])
const scriptJson = value => JSON.stringify(value).replace(/</g, '\\u003c').replace(/\u2028/g, '\\u2028').replace(/\u2029/g, '\\u2029')

function pluginHelp(plugin) {
  const list = typeof plugin.help === 'string' ? [plugin.help] : Array.isArray(plugin.help) ? plugin.help : []
  const help = list.filter(value => typeof value === 'string' && value.trim()).map(value => value.trim())
  if (help.length) return help
  if (typeof plugin.command === 'string') return [plugin.command]
  if (Array.isArray(plugin.command)) return plugin.command.filter(value => typeof value === 'string')
  return []
}

function pluginTags(plugin) {
  const tags = Array.isArray(plugin.tags) ? plugin.tags : typeof plugin.tags === 'string' ? [plugin.tags] : []
  return tags.map(tag => clean(tag).toLowerCase()).filter(Boolean)
}

export function dashboardRegistry(plugins = []) {
  const active = plugins.filter(plugin => plugin && !plugin.disabled)
  const allTags = new Set(active.flatMap(pluginTags))
  const knownTags = new Set(DASHBOARD_GROUPS.flatMap(group => group.tags))
  const otherTags = [...allTags].filter(tag => !knownTags.has(tag)).sort()
  const groups = DASHBOARD_GROUPS.map(group => {
    const availableTags = [...group.tags, ...(group.number === 13 ? otherTags : [])].filter(tag => allTags.has(tag))
    const commands = new Set()
    active.forEach(plugin => {
      if (pluginTags(plugin).some(tag => availableTags.includes(tag))) pluginHelp(plugin).forEach(help => commands.add(help))
    })
    return { ...group, tags: availableTags, count: commands.size }
  })
  return { groups, totalCommands: new Set(active.flatMap(pluginHelp)).size, totalPlugins: active.length }
}

export function dashboardShortcut(command, text, plugins) {
  const match = /^menu(1[0-3]|[1-9])$/i.exec(command || '')
  if (!match) return text || ''
  const group = dashboardRegistry(plugins).groups.find(item => item.number === Number(match[1]))
  // An absent category uses its configured tags so it returns the normal "not found" message.
  return (group?.tags.length ? group.tags : DASHBOARD_GROUPS[Number(match[1]) - 1].tags).join(',')
}

export function dashboardPicker(registry, prefix = '.') {
  return SECTION_TITLES.map((title, index) => ({
    title,
    rows: registry.groups.filter(group => group.section === index && group.count > 0).map(group => ({
      title: group.title,
      description: group.description + ' · ' + group.count + ' cmd',
      id: prefix + 'menu' + group.number
    }))
  })).filter(section => section.rows.length)
}

export async function dashboardDeadline(task, milliseconds, stage) {
  let timer
  try {
    return await Promise.race([task, new Promise((_, reject) => {
      timer = setTimeout(() => reject(new Error(stage + ' timeout')), milliseconds)
    })])
  } finally { clearTimeout(timer) }
}

function weatherDescription(code) {
  if (code === 0) return 'Cerah ☀️'
  if (code === 1) return 'Cerah berawan 🌤️'
  if (code === 2) return 'Berawan sebagian 🌤️'
  if (code === 3) return 'Berawan ☁️'
  if ([45, 48].includes(code)) return 'Berkabut 🌫️'
  if ([51, 53, 55, 56, 57].includes(code)) return 'Gerimis 🌦️'
  if ([61, 63, 65, 66, 67, 80, 81, 82].includes(code)) return 'Hujan 🌧️'
  if ([71, 73, 75, 77, 85, 86].includes(code)) return 'Salju ❄️'
  if ([95, 96, 97, 99].includes(code)) return 'Hujan petir ⛈️'
  return 'Kondisi cuaca'
}

export async function dashboardWeather(city = 'Bandung', { fetchImpl = fetch, timeout = 3500 } = {}) {
  city = clean(city).slice(0, 80) || 'Bandung'
  const key = city.toLowerCase()
  const cached = weatherCache.get(key)
  if (cached && cached.expires > Date.now()) return cached.value
  const missing = { city, text: 'Cuaca belum tersedia', available: false }
  try {
    const signal = AbortSignal.timeout(timeout)
    const value = await dashboardDeadline((async () => {
      const locationUrl = new URL('https://geocoding-api.open-meteo.com/v1/search')
      locationUrl.searchParams.set('name', city)
      locationUrl.searchParams.set('count', '1')
      locationUrl.searchParams.set('language', 'id')
      locationUrl.searchParams.set('countryCode', 'ID')
      const locationResponse = await fetchImpl(locationUrl, { signal })
      if (!locationResponse.ok) throw new Error('Location unavailable')
      const location = (await locationResponse.json()).results?.[0]
      if (!Number.isFinite(location?.latitude) || !Number.isFinite(location?.longitude)) throw new Error('City not found')
      const weatherUrl = new URL('https://api.open-meteo.com/v1/forecast')
      weatherUrl.searchParams.set('latitude', String(location.latitude))
      weatherUrl.searchParams.set('longitude', String(location.longitude))
      weatherUrl.searchParams.set('current', 'temperature_2m,weather_code')
      weatherUrl.searchParams.set('timezone', 'Asia/Jakarta')
      const weatherResponse = await fetchImpl(weatherUrl, { signal })
      if (!weatherResponse.ok) throw new Error('Weather unavailable')
      const current = (await weatherResponse.json()).current
      if (!Number.isFinite(current?.temperature_2m) || !Number.isFinite(current?.weather_code)) throw new Error('Weather incomplete')
      return { city: clean(location.name || city), text: weatherDescription(current.weather_code) + ' ' + Math.round(current.temperature_2m) + '°C', available: true }
    })(), timeout, 'Weather')
    weatherCache.set(key, { value, expires: Date.now() + 600000 })
    while (weatherCache.size > 8) weatherCache.delete(weatherCache.keys().next().value)
    return value
  } catch {
    weatherCache.set(key, { value: missing, expires: Date.now() + 60000 })
    return missing
  }
}

async function dashboardBanner() {
  if (cachedBanner !== null) return cachedBanner
  for (const file of ['menu-v7-banner.gif', 'thumbnail.jpg', 'banner.jpg']) {
    try {
      const bytes = await readFile(new URL('../media/' + file, import.meta.url))
      if (bytes.length > 250000) continue
      const type = file.endsWith('.gif') ? 'image/gif' : 'image/jpeg'
      cachedBanner = 'data:' + type + ';base64,' + bytes.toString('base64')
      return cachedBanner
    } catch {}
  }
  return ''
}

function dashboardTips(plugins, prefix) {
  const help = new Set(plugins.filter(plugin => !pluginTags(plugin).includes('owner')).flatMap(pluginHelp))
  const tips = []
  for (const keyword of ['play', 'igstalkcard', 'brat', 'sticker', 'ping', 'owner']) {
    const example = [...help].find(item => item.split(/\s+/)[0].toLowerCase() === keyword)
    if (example) tips.push('💡 Coba ' + prefix + example)
  }
  tips.push('📂 Pilih kategori lewat tombol BUKA MENU.')
  tips.push('✨ Ketik ' + prefix + 'menu all untuk melihat semua perintah.')
  return tips
}

export function dashboardRichContent(data) {
  const responseId = 'jere-dashboard-' + Date.now().toString(36) + '-' + Math.random().toString(36).slice(2, 10)
  const caption = '✨ *' + data.botName + '*\n👋 Halo, ' + data.userName + '!\n\nStatus: ' + data.role + '\nMode: ' + data.mode + '\nTotal fitur: ' + data.totalCommands + '\n\nTekan ✦ BUKA MENU di pesan berikutnya untuk memilih kategori.'
  return {
    messageContextInfo: { deviceListMetadata: {}, deviceListMetadataVersion: 2, botMetadata: { botResponseId: responseId } },
    botForwardedMessage: { message: { richResponseMessage: {
      messageType: 1,
      submessages: [{ messageType: 2, messageText: caption }],
      unifiedResponse: { data: Buffer.from(JSON.stringify({ response_id: responseId, sections: [{
        __typename: 'GenAIUnifiedResponseSection',
        view_model: { __typename: 'GenAISingleLayoutViewModel', primitive: { __typename: 'GenAIaeacdsnwHtmlPrimitive', payload: dashboardHtml(data), trusted_sources: [] } }
      }] })) },
      contextInfo: { isForwarded: true, forwardingScore: 1, forwardOrigin: 4 }
    } } }
  }
}

export async function sendDashboardMenu(m, conn, caption, options = {}) {
  const prefix = options.usedPrefix || '.'
  const plugins = options.plugins || []
  const registry = dashboardRegistry(plugins)
  const setting = global.db?.data?.settings?.[conn.user?.jid || conn.user?.id] || {}
  const botName = clean(options.botInfo?.name || global.botname || 'Jere-MD').slice(0, 80)
  const fallbackIndex = '📂 *' + botName + ' — KATEGORI MENU*\n\n' + registry.groups.filter(group => group.count > 0).map(group => prefix + 'menu' + group.number + ' • ' + group.title + ' (' + group.count + ' cmd)').join('\n')
  if (!options.query) {
    const status = clean(options.user?.status)
    const data = {
      botName, prefix,
      userName: clean(options.user?.name || m.pushName || 'User').slice(0, 60),
      role: options.isOwner || options.isROwner ? '👑 Owner' : /vip/i.test(status) ? '💎 VIP' : /premium/i.test(status) ? '⭐ Premium' : '👤 User',
      mode: setting.gconly || global.opts?.gconly ? 'Group' : global.opts?.self || conn.public === false ? 'Self' : 'Public',
      totalCommands: registry.totalCommands,
      uptime: process.uptime(),
      generatedAt: Date.now(),
      weather: await dashboardWeather(setting.menuCity || global.menuCity || 'Bandung', { fetchImpl: options.fetchImpl || fetch, timeout: options.weatherTimeout || 3500 }),
      banner: options.banner ?? await dashboardBanner(),
      tips: dashboardTips(plugins, prefix)
    }
    try {
      await dashboardDeadline((async () => {
        const generate = options.generate || (await import('@adiwajshing/baileys')).generateWAMessageFromContent
        const message = generate(m.chat, dashboardRichContent(data), { userJid: conn.user?.jid || conn.user?.id, quoted: m })
        return conn.relayMessage(m.chat, message.message, { messageId: message.key.id })
      })(), options.sendTimeout || 7000, 'Dashboard')
    } catch {
      await dashboardDeadline(m.reply('✨ *' + botName + '*\n👋 Halo, ' + data.userName + '!\nStatus: ' + data.role + ' • Mode: ' + data.mode + '\nTotal fitur: ' + data.totalCommands + '\n📍 ' + data.weather.city + ' • ' + data.weather.text), options.sendTimeout || 7000, 'Dashboard text').catch(() => {})
    }
  }
  const sections = dashboardPicker(registry, prefix)
  const body = options.query ? caption.replace(/\u200e/g, '') : '📂 *Pilih kategori, atau ketik ' + prefix + 'menu1 – ' + prefix + 'menu13*'
  try {
    await dashboardDeadline(conn.sendButton(m.chat, {
      caption: body,
      footer: botName + ' • Explore your commands',
      buttons: [{ name: 'single_select', buttonParamsJson: JSON.stringify({ title: '✦ BUKA MENU', sections }) }]
    }, { quoted: m }), options.sendTimeout || 7000, 'Category picker')
  } catch {
    await dashboardDeadline(m.reply(options.query ? body + '\n\n' + fallbackIndex : fallbackIndex), options.sendTimeout || 7000, 'Category text')
  }
}
